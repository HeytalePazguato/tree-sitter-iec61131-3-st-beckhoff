================================================================================
PROGRAM with IF and FOR
================================================================================

PROGRAM Main
VAR
    i : INT;
    x : INT;
END_VAR
IF x > 0 THEN
    x := x - 1;
ELSE
    x := 0;
END_IF
FOR i := 0 TO 9 DO
    x := x + i;
END_FOR
END_PROGRAM
